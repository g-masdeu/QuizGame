class _WrongOverlay extends StatelessWidget {
  final double opacity;
  const _WrongOverlay({required this.opacity});

  @override
  Widget build(BuildContext context) {
    final size = MediaQuery.of(context).size;
    final animSide = (size.shortestSide * 0.2).clamp(220.0, 520.0);

    final stack = Stack(
      children: [
        // Fondo con gradiente + blur (tono cálido moderno)
        Container(
          decoration: BoxDecoration(
            gradient: RadialGradient(
              colors: [
                Colors.amberAccent.withOpacity(0.90),
                const ui.Color.fromARGB(255, 255, 109, 64).withOpacity(0.95),
                Colors.black.withOpacity(0.92),
              ],
              center: Alignment.center,
              radius: 2.0,
            ),
          ),
          child: BackdropFilter(
            filter: ui.ImageFilter.blur(sigmaX: 10, sigmaY: 10),
            child: const SizedBox.expand(),
          ),
        ),

        // Contenido central: animación + texto
        Center(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              // Lottie de fallo (centrada y escalable)
              SizedBox(
                width: animSide,
                height: animSide,
                child: Lottie.asset(
                  'assets/anim/error.json',
                  repeat: true,
                  frameRate: FrameRate.max,
                  fit: BoxFit.contain,
                ),
              ),
              const SizedBox(height: 24),
              Text(
                '¡INTÉNTALO DE NUEVO!',
                textAlign: TextAlign.center,
                style: TextStyle(
                  fontSize: 48,
                  fontWeight: FontWeight.w900,
                  color: Colors.white,
                  letterSpacing: 2,
                  shadows: [
                    Shadow(
                      color: Colors.black.withOpacity(0.8),
                      blurRadius: 20,
                      offset: const Offset(0, 6),
                    ),
                  ],
                ),
              ),
            ],
          ),
        ),
      ],
    );

    return IgnorePointer(
      ignoring: true,
      child: AnimatedOpacity(
        opacity: opacity,
        duration: const Duration(milliseconds: 200),
        child: stack,
      ),
    );
  }
}

Future<void> _showCorrectAndProceed() async {
    HapticFeedback.heavyImpact();
    setState(() {
      _showCorrectOverlay = true;
      _overlayOpacity = 0.0;
      _wrongIndices.clear();
    });

    await Future.delayed(const Duration(milliseconds: 10));
    setState(() => _overlayOpacity = 1.0);
    _confetti.play();

    await Future.delayed(const Duration(milliseconds: 2000));

    setState(() => _overlayOpacity = 0.0);
    await Future.delayed(const Duration(milliseconds: 200));

    if (!mounted) return;
    setState(() => _showCorrectOverlay = false);

    _loadNextQuestion();
  }
